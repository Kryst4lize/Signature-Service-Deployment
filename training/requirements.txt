# Training environment.
#
# GPU note: torch here is the CUDA build, which pulls ~1.7 GB of nvidia-* wheels.
# It is needed only for the CycleGAN training and the .pth -> ONNX export. If you
# train CycleGAN elsewhere and only run the Keras stages, install the CPU wheel
# instead and drop ~1.7 GB from the image:
#     pip install torch==2.3.0 --index-url https://download.pytorch.org/whl/cpu

# ── Deep learning ─────────────────────────────────────────────────────────────
tensorflow==2.21.0
torch==2.3.0
torchvision==0.18.0

# ── CycleGAN (upstream repo imports these unconditionally) ────────────────────
# util/visualizer.py has a module-level `import wandb`; util/html.py imports
# dominate. They look unused because the repo is cloned, not vendored.
wandb==0.26.1
dominate==2.9.1

# ── Detector export ───────────────────────────────────────────────────────────
# Was missing entirely: the YOLO export step was documented but unimplemented,
# and both old converters imported ultralytics without declaring it.
ultralytics==8.3.49

# ── ONNX ──────────────────────────────────────────────────────────────────────
onnx==1.17.0
# 1.17.0, not 1.16.1: the older release pins protobuf~=3.20 while
# tensorflow 2.21.0 requires protobuf>=6.31.1, which is unsatisfiable.
# 1.17.0 relaxes it to protobuf>=3.20 and adds the Keras 3 compatibility
# shims tf2onnx.convert.from_keras needs against this TensorFlow.
tf2onnx==1.17.0
# Graph simplification. The old converter imported `onnxsim` while this file
# pinned `onnxslim` — different projects — so a clean install died with
# ModuleNotFoundError before converting anything. Resolved in favour of
# onnxslim: it is pure Python with wheels for every platform, whereas onnxsim
# ships no aarch64 wheel and needs cmake plus a C++ toolchain to build.
onnxslim==0.1.96

# ── Data & numerics ───────────────────────────────────────────────────────────
numpy==1.26.4
scipy==1.14.1
scikit-learn==1.5.2
pillow==11.0.0
# headless: nothing here calls imshow/namedWindow/waitKey, and the GUI build
# fails to import in a slim container for want of libGL.
opencv-python-headless==4.10.0.84
matplotlib==3.9.3
tqdm==4.67.1
PyYAML==6.0.2

# ── Experiment tracking ───────────────────────────────────────────────────────
tensorboard==2.18.0

# Removed:
#   visdom==0.2.4   Not imported anywhere upstream — it survives only in
#                   comments and --help strings, and upstream's environment.yml
#                   lists just dominate/Pillow/wandb. Its legacy setup.py also
#                   imports pkg_resources without declaring setuptools, so it
#                   fails to build against current setuptools:
#                     ModuleNotFoundError: No module named 'pkg_resources'
#   triton==2.3.0   This is openai/triton, the GPU kernel DSL - unrelated to
#                   tritonclient. torch already declares it, guarded by
#                   `platform_system == "Linux" and platform_machine == "x86_64"
#                   and python_version < "3.12"`. Pinning it unconditionally
#                   made `pip install -r requirements.txt` fail outright on
#                   macOS, aarch64 and Python 3.12+, because it publishes only
#                   manylinux x86_64 wheels and no sdist.
#   protobuf==7.34.1  Let tensorflow/onnx resolve a mutually compatible version.
#   setuptools        Build-time only; declared in pyproject.toml.
#   python-dotenv     Configuration is configs/*.yaml now.
